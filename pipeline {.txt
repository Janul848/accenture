pipeline {
    agent any

    stages {
        stage('Checkout') {
            steps {
                // Checkout the code from the repository
                git branch: "${env.BRANCH_NAME}", url: 'https://github.com/YOUR_USERNAME/mock-company-webapp.git'
            }
        }

        stage('Build') {
            steps {
                // Build the application using Gradle
                sh './gradlew build'
            }
        }

        stage('Test') {
            steps {
                // Run the tests
                sh './gradlew test'
            }
        }
    }

    post {
        success {
            echo 'Build and tests succeeded!'
        }
        failure {
            echo 'Build or tests failed!'
        }
    }
}
